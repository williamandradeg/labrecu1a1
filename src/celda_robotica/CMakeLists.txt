cmake_minimum_required(VERSION 3.16)
project(celda_robotica)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(CellEnvironment SHARED src/CellEnvironment.cc)
target_compile_features(CellEnvironment PRIVATE cxx_std_17)
target_link_libraries(CellEnvironment gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS CellEnvironment LIBRARY DESTINATION lib)
install(DIRECTORY launch config worlds urdf meshes DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/cycle.py DESTINATION lib/${PROJECT_NAME})
ament_package()
